# Experiment No. 4
# One-Dimensional Array Operations
# Insertion, Deletion, Traversal and Searching


# Function for Insertion
def insert_element(arr, element, position):
    if position < 0 or position > len(arr):
        print("Invalid position!")
    else:
        arr.insert(position, element)
        print("Element inserted successfully.")


# Function for Deletion
def delete_element(arr, position):
    if position < 0 or position >= len(arr):
        print("Invalid position!")
    else:
        deleted = arr.pop(position)
        print("Deleted element:", deleted)


# Function for Traversal
def traverse(arr):
    print("Array elements:")
    for i in range(len(arr)):
        print(arr[i], end=" ")
    print()


# Function for Linear Search
def search_element(arr, element):
    for i in range(len(arr)):
        if arr[i] == element:
            return i

    return -1


# Main Program

# Create an array
arr = []

# Input number of elements
n = int(input("Enter number of elements: "))

# Read elements
for i in range(n):
    value = int(input(f"Enter element {i + 1}: "))
    arr.append(value)

print("\nOriginal Array:")
print(arr)


# Insertion
element = int(input("\nEnter element to insert: "))
position = int(input("Enter position for insertion (0-based index): "))

insert_element(arr, element, position)

print("Array after insertion:")
print(arr)


# Deletion
position = int(input("\nEnter position to delete (0-based index): "))

delete_element(arr, position)

print("Array after deletion:")
print(arr)


# Traversal
print("\nTraversal:")
traverse(arr)


# Searching
element = int(input("\nEnter element to search: "))

result = search_element(arr, element)

if result != -1:
    print(f"Element {element} found at index {result}.")
else:
    print(f"Element {element} not found in the array.")
