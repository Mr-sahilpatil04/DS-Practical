# Radix Sort for Student Percentages

# Counting Sort function
def counting_sort(arr, exp):
    n = len(arr)
    output = [0] * n
    count = [0] * 10

    # Count occurrences of digits
    for i in range(n):
        digit = (arr[i] // exp) % 10
        count[digit] += 1

    # Calculate cumulative count
    for i in range(1, 10):
        count[i] += count[i - 1]

    # Build output array
    for i in range(n - 1, -1, -1):
        digit = (arr[i] // exp) % 10
        output[count[digit] - 1] = arr[i]
        count[digit] -= 1

    # Copy output to original array
    for i in range(n):
        arr[i] = output[i]


# Radix Sort function
def radix_sort(arr):
    if len(arr) == 0:
        return arr

    # Find maximum element
    max_value = max(arr)

    # Start with units digit
    exp = 1

    # Process each digit
    while max_value // exp > 0:
        counting_sort(arr, exp)
        exp *= 10

    return arr


# Main Program

# Input number of students
n = int(input("Enter number of students: "))

# Store percentages
percentages = []

for i in range(n):
    percentage = float(input(f"Enter percentage of student {i + 1}: "))
    percentages.append(percentage)

print("\nOriginal Percentages:")
print(percentages)


# Convert floating-point percentages to integers
# Multiply by 100 to preserve two decimal places
integer_percentages = [int(round(x * 100)) for x in percentages]

# Apply Radix Sort
radix_sort(integer_percentages)

# Convert back to percentage format
sorted_percentages = [x / 100 for x in integer_percentages]

print("\nSorted Percentages:")
print(sorted_percentages)


# Display Top 5 Scores
top_five = sorted_percentages[-5:][::-1]

print("\nTop 5 Highest Scores:")

for i, score in enumerate(top_five, 1):
    print(f"{i}. {score:.2f}%")
